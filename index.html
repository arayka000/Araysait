const TEACHER_PASSWORD = "infonet6"; // Мұғалім құпиясөзі (өзгертуге болады) const SHEET_NAME = "Нәтижелер"; const HEADERS = [ "Уақыты", "Аты-жөні", "Сыныбы", "Дескрипторлар", "Өзін-өзі бағалау", "Деңгей", "Бүгінгі жетістігім", "Даму қадамым", "Мұғалім балы", "Мұғалім пікірі" ]; function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Кестеге формула ретінде орындалып кетпеуі үшін (=, +, -, @ деп басталса) function clean_(v) { if (v === null || v === undefined) return ""; v = String(v).slice(0, 1000); return /^[=+\-@]/.test(v) ? "'" + v : v; } function cleanScore_(v) { if (v === "" || v === null || v === undefined) return ""; const n = Number(v); if (isNaN(n)) return ""; return Math.max(0, Math.min(10, n)); } function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); } return sheet; } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(15000); // бір уақытта көп оқушы жіберсе, жолдар араласпайды const data = JSON.parse(e.postData.contents); const sheet = getSheet_(); // 1. Оқушы дерегін сақтау if (data.action === "submitStudent") { const name = clean_(data.name).trim(); const studentClass = clean_(data.studentClass).trim(); if (!name || !studentClass) { return json_({ status: "error", message: "Аты-жөні мен сыныбын толтырыңыз!" }); } // Баллды клиенттен емес, дескрипторлардан сервер өзі есептейді let descriptors = Array.isArray(data.descriptors) ? data.descriptors.slice(0, 10) : []; descriptors = descriptors.map(function (d) { return d ? 1 : 0; }); const selfScore = descriptors.reduce(function (a, b) { return a + b; }, 0); const level = selfScore >= 9 ? "Жоғары" : (selfScore >= 5 ? "Орта" : "Төмен"); sheet.appendRow([ new Date().toLocaleString("kk-KZ"), name, studentClass, JSON.stringify(descriptors), selfScore, level, clean_(data.reflection1), clean_(data.reflection2), "", "" ]); return json_({ status: "success" }); } // 2. Мұғалімнің кіруі және деректерді алуы if (data.action === "teacherLogin") { if (data.password !== TEACHER_PASSWORD) { return json_({ status: "error", message: "Құпиясөз қате!" }); } const rows = sheet.getDataRange().getValues(); rows.shift(); // тақырыпты алып тастау const students = rows.map(function (row, index) { return { rowIndex: index + 2, timestamp: row[0], name: row[1], studentClass: row[2], descriptors: row[3], selfScore: row[4], level: row[5], reflection1: row[6], reflection2: row[7], teacherScore: row[8], teacherFeedback: row[9] }; }); return json_({ status: "success", students: students }); } // 3. Мұғалім бағасы мен пікірін жаңарту if (data.action === "updateScore") { if (data.password !== TEACHER_PASSWORD) { return json_({ status: "error", message: "Құпиясөз қате!" }); } const rowIndex = Number(data.rowIndex); if (!Number.isInteger(rowIndex) || rowIndex < 2 || rowIndex > sheet.getLastRow()) { return json_({ status: "error", message: "Жол нөмірі қате!" }); } sheet.getRange(rowIndex, 9).setValue(cleanScore_(data.teacherScore)); sheet.getRange(rowIndex, 10).setValue(clean_(data.teacherFeedback)); return json_({ status: "success" }); } // 4. Оқушыны қолмен қосу if (data.action === "addStudent") { if (data.password !== TEACHER_PASSWORD) { return json_({ status: "error", message: "Құпиясөз қате!" }); } sheet.appendRow([ new Date().toLocaleString("kk-KZ"), clean_(data.name), clean_(data.studentClass), "[]", 0, "Төмен", "", "", cleanScore_(data.teacherScore), clean_(data.teacherFeedback) ]); return json_({ status: "success" }); } return json_({ status: "error", message: "Белгісіз әрекет" }); } catch (err) { return json_({ status: "error", message: err.toString() }); } finally { try { lock.releaseLock(); } catch (e2) {} } } function doGet(e) { return ContentService.createTextOutput("Информатика бағалау жүйесінің бэкенді сәтті жұмыс істеп тұр!"); }
